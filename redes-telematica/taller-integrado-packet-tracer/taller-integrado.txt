===============================================================================
 PARTE 6 - TELEFONIA IP (VOZ IP)   en LAN5 (R5) y LAN6 (R6)
===============================================================================
 Se usa DHCP con "option 150" (apunta al TFTP = el propio router-gateway),
 telephony-service (CME) y dial-peer voip para llamadas entre las dos LAN.
 Plan de numeracion:  LAN5 -> 51xx     LAN6 -> 61xx

--------- R5 : DHCP + servicio de voz ----------------------------------------
configure terminal
ip dhcp pool VOZ5
 network 192.168.10.64 255.255.255.192
 default-router 192.168.10.65
 option 150 ip 192.168.10.65
 exit
telephony-service
 max-ephones 5
 max-dn 5
 ip source-address 192.168.10.65 port 2000
 auto assign 1 to 5
 exit
ephone-dn 1
 number 5101
 exit
ephone-dn 2
 number 5102
 exit
exit
wr

--------- SW5 (switch de voz, puertos donde van los telefonos) ---------------
configure terminal
interface range fastethernet 0/1 - 5
 switchport voice vlan 1
 exit
exit
wr

--------- R6 : DHCP + servicio de voz ----------------------------------------
configure terminal
ip dhcp pool VOZ6
 network 192.168.10.160 255.255.255.224
 default-router 192.168.10.161
 option 150 ip 192.168.10.161
 exit
telephony-service
 max-ephones 5
 max-dn 5
 ip source-address 192.168.10.161 port 2000
 auto assign 1 to 5
 exit
ephone-dn 1
 number 6101
 exit
ephone-dn 2
 number 6102
 exit
exit
wr

--------- SW6 (switch de voz) ------------------------------------------------
configure terminal
interface range fastethernet 0/1 - 5
 switchport voice vlan 1
 exit
exit
wr

--------- DIAL-PEER: llamadas ENTRE las dos LAN ------------------------------
! En R5: las llamadas a numeros 6... salen hacia el gateway de LAN6 (.161)
configure terminal
dial-peer voice 1 voip
 destination-pattern 6...
 session target ipv4:192.168.10.161
 exit
exit
wr

! En R6: las llamadas a numeros 5... salen hacia el gateway de LAN5 (.65)
configure terminal
dial-peer voice 1 voip
 destination-pattern 5...
 session target ipv4:192.168.10.65
 exit
exit
wr
